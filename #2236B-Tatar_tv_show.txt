#include <bits/stdc++.h>
using namespace std;

int main(){
    int t;
    scanf("%d", &t);
    while(t--){
        int n, k;
        scanf("%d %d", &n, &k);
        static char s[200005];
        scanf("%s", s);

        vector<int> parity(k, 0);
        for(int i = 0; i < n; i++){
            if(s[i] == '1') parity[i % k] ^= 1;
        }

        bool ok = true;
        for(int i = 0; i < k; i++){
            if(parity[i]){ ok = false; break; }
        }

        puts(ok ? "YES" : "NO");
    }
    return 0;
}
