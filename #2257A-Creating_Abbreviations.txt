#2257A-Creating_Abbreviations
#include <bits/stdc++.h>
using namespace std;

int main(){
    int t;
    scanf("%d", &t);
    while(t--){
        int n, m;
        scanf("%d %d", &n, &m);
        bool avail[26] = {false};
        for(int i = 0; i < n; i++){
            char buf[25];
            scanf("%s", buf);
            avail[buf[0]-'a'] = true;
        }
        vector<string> abbr(m);
        for(int i = 0; i < m; i++){
            char buf[25];
            scanf("%s", buf);
            abbr[i] = string(buf);
        }

        vector<bool> done(m, false);
        int doneCount = 0;
        bool progress = true;
        while(progress && doneCount < m){
            progress = false;
            for(int i = 0; i < m; i++){
                if(done[i]) continue;
                bool ok = true;
                for(char c : abbr[i]){
                    if(!avail[c-'A']){ ok = false; break; }
                }
                if(ok){
                    done[i] = true;
                    doneCount++;
                    avail[abbr[i][0]-'A'] = true;
                    progress = true;
                }
            }
        }

        printf(doneCount == m ? "YES\n" : "NO\n");
    }
    return 0;
}
