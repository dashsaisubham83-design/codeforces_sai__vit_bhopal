#include <bits/stdc++.h>
using namespace std;

int main(){
    int t;
    scanf("%d", &t);
    while(t--){
        char buf[300005];
        scanf("%s", buf);
        string s(buf);
        int n = s.size();

        // remove all 4's, keep the rest (1,2,3) in order
        string t13;
        int fours = 0;
        for(char c : s){
            if(c == '4') fours++;
            else t13.push_back(c);
        }
        int m = t13.size();

        // prefix2count[i] = number of '2' in t13[0..i-1]
        vector<int> prefix2(m+1, 0);
        for(int i = 0; i < m; i++){
            prefix2[i+1] = prefix2[i] + (t13[i]=='2' ? 1 : 0);
        }
        // suffix13[i] = number of '1' or '3' in t13[i..m-1]
        vector<int> suffix13(m+1, 0);
        for(int i = m-1; i >= 0; i--){
            suffix13[i] = suffix13[i+1] + ((t13[i]=='1'||t13[i]=='3') ? 1 : 0);
        }

        int best = 0;
        for(int i = 0; i <= m; i++){
            best = max(best, prefix2[i] + suffix13[i]);
        }

        int removals = fours + (m - best);
        printf("%d\n", removals);
    }
    return 0;
}
