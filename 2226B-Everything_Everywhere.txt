#include <bits/stdc++.h>
using namespace std;

int main(){
    int t;
    scanf("%d", &t);
    while(t--){
        int n;
        scanf("%d", &n);
        vector<long long> p(n);
        for(auto &x : p) scanf("%lld", &x);

        long long count = 0;
        for(int i = 0; i+1 < n; i++){
            long long x = min(p[i], p[i+1]);
            long long d = llabs(p[i]-p[i+1]);
            if(x % d == 0) count++;
        }

        printf("%lld\n", count);
    }
    return 0;
}
