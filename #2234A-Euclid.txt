#include <bits/stdc++.h>
using namespace std;

int main(){
    int t;
    scanf("%d", &t);
    while(t--){
        int n;
        scanf("%d", &n);
        vector<long long> b(n);
        for(auto &x : b) scanf("%lld", &x);
        sort(b.rbegin(), b.rend()); // descending

        bool ok = true;
        for(int i = 0; i < n-2; i++){
            if(b[i+2] != b[i] % b[i+1]){ ok = false; break; }
        }
        if(ok) printf("%lld %lld\n", b[0], b[1]);
        else printf("-1\n");
    }
    return 0;
}
