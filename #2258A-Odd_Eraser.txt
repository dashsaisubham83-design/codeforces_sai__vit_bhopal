#2258A-Odd_Eraser
#include <bits/stdc++.h>
using namespace std;

int main(){
    int t;
    scanf("%d", &t);
    while(t--){
        int n;
        scanf("%d", &n);
        vector<long long> a(n);
        for(auto &x : a) scanf("%lld", &x);
        long long ans = __gcd(a[0], a[n-1]);
        printf("%lld\n", ans);
    }
    return 0;
}
