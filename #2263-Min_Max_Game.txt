#2263-Min_Max_Game
#include <bits/stdc++.h>
using namespace std;

int main(){
    int t;
    scanf("%d", &t);
    while(t--){
        int n;
        scanf("%d", &n);
        int ones = 0;
        for(int i = 0; i < n; i++){
            int x;
            scanf("%d", &x);
            ones += x;
        }
        int need = (n + 1) / 2; // ceil(n/2)
        puts(ones >= need ? "Bessie" : "Elsie");
    }
}
