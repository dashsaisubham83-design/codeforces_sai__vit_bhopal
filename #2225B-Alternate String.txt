#2225B-Alternate String

#include <bits/stdc++.h>
using namespace std;

int main(){
    int t;
    scanf("%d", &t);
    while(t--){
        char buf[200005];
        scanf("%s", buf);
        int n = strlen(buf);
        int defects = 0;
        for(int i = 0; i + 1 < n; i++){
            if(buf[i] == buf[i+1]) defects++;
        }
        printf(defects <= 2 ? "YES\n" : "NO\n");
    }
    return 0;
}
