#include <bits/stdc++.h>
using namespace std;

int main(){
    int t;
    scanf("%d", &t);
    while(t--){
        int n;
        scanf("%d", &n);
        vector<int> h(n);
        for(auto &x : h) scanf("%d", &x);
        int mx = *max_element(h.begin(), h.end());
        int mn = *min_element(h.begin(), h.end());
        printf("%d\n", mx - mn + 1);
    }
    return 0;
}
