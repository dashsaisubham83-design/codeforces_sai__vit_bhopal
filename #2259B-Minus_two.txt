#2259B-Minus_two

#include <bits/stdc++.h>
using namespace std;

int main(){
    int t;
    scanf("%d", &t);
    while (t--) {
        int n;
        scanf("%d", &n);

        long long oddCount = 0, mod0 = 0, mod2 = 0; // mod0: a%4==0, mod2: a%4==2

        for (int i = 0; i < n; i++) {
            long long a;
            scanf("%lld", &a);
            if (a % 2 != 0) {
                // odd numbers all eventually converge to the fixed point 1
                oddCount++;
            } else {
                // even numbers eventually oscillate between 0 and 2;
                // which one they land on (for large k) depends on (a/2) mod 2,
                // equivalently a mod 4
                if (a % 4 == 0) mod0++;
                else mod2++;
            }
        }

        long long ans = max(oddCount, max(mod0, mod2));
        printf("%lld\n", ans);
    }
    return 0;
}
